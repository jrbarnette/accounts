--
Data

Account list (account store) widget (JList; want to be JTree)
    selected account
    event: new selection
account data index
Account fields (AccountDataPanel)
    collection of JTextField elements with account data
    event: changed field

edit state (CHANGED, VALID) <
    JTextField contents

edit state (EXISTING, EARLIEST, LATEST) <
    selected account
    account data index

--
select account -> ListSelectionListener valueChanged
    state: not CHANGED
    set account data index to 0
    update account data fields from selected account data
	-> DocumentListener insertUpdate, removeUpdate
	state -> EXISTING, not CHANGED, VALID, LATEST

earlier history -> ActionListener actionPerformed: EARLIER
    state: EXISTING, not CHANGED, not EARLIEST
    increment account data index
    update account data fields from selected account data
	-> DocumentListener insertUpdate, removeUpdate
	state -> EXISTING, not CHANGED, VALID, not LATEST

later history -> ActionListener actionPerformed: LATER
    state: EXISTING, not CHANGED, not LATEST
    decrement account data index
    update account data fields from selected account data
	-> DocumentListener insertUpdate, removeUpdate
	state -> EXISTING, not CHANGED, VALID, not EARLIEST

new account (clear fields) -> ActionListener actionPerformed: CLEAR
    state: not CHANGED
    clear selected account in account list
	-> ListSelectionListener valueChanged

delete account -> ActionListener actionPerformed: DELETE
    assert: an account is selected
    assert: no fields have changed
    get selected account from account list
    delete selected account from account store
    update account list from account store
	(implicitly clear list selection)
	-> ListSelectionListener valueChanged

update account -> ActionListener actionPerformed: UPDATE
    assert: an account is selected
    assert: some fields have changed
    assert: all fields are valid
    get selected account from account list
    update selected account in account store with account data from fields
	(text for "latest change" will be updated)

add account -> ActionListener actionPerformed: ADD
    assert: no selected account
    assert: some fields have changed
    assert: all fields are valid
    create account in account store with account data from fields
    update account list from account store
	(update list selection to new account)
	-> ListSelectionListener valueChanged

revert edits -> ActionListener actionPerformed: REVERT
    assert: some fields have changed
    restore account data fields from selected account
	-> DocumentListener insertUpdate, removeUpdate

copy password -> ActionListener actionPerformed: COPY
    assert: an account is selected
    assert: no fields have changed
    copy contents of password field
	(or password from selected account data)
    (no changes)

generate password -> ActionListener actionPerformed: GENERATE
    set password field to generated content
    -> DocumentListener insertUpdate, removeUpdate

edit field -> DocumentListener insertUpdate, removeUpdate
    update button/control states from field states
